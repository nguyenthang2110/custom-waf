% Cấu hình listings cho các ngôn ngữ Go, Python, JSON, YAML, SQL, Bash
\definecolor{codegreen}{rgb}{0.16,0.55,0.26}
\definecolor{codegray}{rgb}{0.45,0.45,0.45}
\definecolor{codepurple}{rgb}{0.50,0.10,0.50}
\definecolor{codeorange}{rgb}{0.85,0.40,0.10}
\definecolor{codebg}{rgb}{0.97,0.97,0.97}
\definecolor{codeframe}{rgb}{0.80,0.80,0.80}

\lstdefinestyle{thesis}{
    backgroundcolor   = \color{codebg},
    commentstyle      = \color{codegreen}\itshape,
    keywordstyle      = \color{codepurple}\bfseries,
    numberstyle       = \tiny\color{codegray},
    stringstyle       = \color{codeorange},
    basicstyle        = \ttfamily\footnotesize,
    breakatwhitespace = false,
    breaklines        = true,
    captionpos        = b,
    keepspaces        = true,
    numbers           = left,
    numbersep         = 6pt,
    showspaces        = false,
    showstringspaces  = false,
    showtabs          = false,
    tabsize           = 2,
    frame             = single,
    rulecolor         = \color{codeframe},
    xleftmargin       = 12pt,
    xrightmargin      = 4pt,
    aboveskip         = 8pt,
    belowskip         = 4pt
}

\lstdefinelanguage{Go}{
    keywords={break,default,func,interface,select,case,defer,go,map,struct,
             chan,else,goto,package,switch,const,fallthrough,if,range,type,
             continue,for,import,return,var,nil,true,false,iota},
    keywordstyle=\color{codepurple}\bfseries,
    ndkeywords={string,int,int8,int16,int32,int64,uint,uint8,uint16,uint32,
               uint64,uintptr,byte,rune,float32,float64,complex64,complex128,
               bool,error},
    ndkeywordstyle=\color{codepurple}\bfseries,
    sensitive=true,
    comment=[l]{//},
    morecomment=[s]{/*}{*/},
    morestring=[b]",
    morestring=[b]`
}

\lstset{style=thesis}
